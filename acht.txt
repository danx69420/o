#include <stdio.h>
#include <stdlib.h>

void sort(int a[], int n) {
    int i, j, temp;

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }
}

int main() {
    int n, head, cylinders;
    int req[100], a[100];
    int i, j, pos;
    int current, total;
    int left[100], right[100];
    int l = 0, r = 0;

    printf("Enter number of disk requests: ");
    scanf("%d", &n);

    if (n < 1 || n > 100) {
        printf("Invalid number of requests.\n");
        return 1;
    }

    printf("Enter disk requests:\n");
    for (i = 0; i < n; i++) {
        scanf("%d", &req[i]);
    }

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("Enter total number of cylinders: ");
    scanf("%d", &cylinders);

    if (cylinders < 1 || head < 0 || head >= cylinders) {
        printf("Invalid disk size or head position.\n");
        return 1;
    }

    for (i = 0; i < n; i++) {
        if (req[i] < 0 || req[i] >= cylinders) {
            printf("Invalid request: %d\n", req[i]);
            return 1;
        }
        a[i] = req[i];
    }

    sort(a, n);

    // Divide requests around the initial head position.
    // The initial direction is toward cylinder 0.

    for (i = 0; i < n; i++) {
        if (a[i] <= head)
            left[l++] = a[i];
        else
            right[r++] = a[i];
    }

    // ---------------- SSTF ----------------

    printf("\nSSTF:\n");

    int visited[100] = {0};
    current = head;
    total = 0;

    printf("Service order: ");

    for (i = 0; i < n; i++) {
        int best = -1;
        int distance = 1000000000;

        for (j = 0; j < n; j++) {
            if (!visited[j]) {
                int d = abs(req[j] - current);

                if (d < distance) {
                    distance = d;
                    best = j;
                }
            }
        }

        visited[best] = 1;
        total += abs(req[best] - current);
        current = req[best];

        printf("%d ", current);
    }

    printf("\nTotal head movement = %d\n", total);

    // ---------------- SCAN ----------------

    printf("\nSCAN (moving toward cylinder 0 first):\n");

    current = head;
    total = 0;

    printf("Service order: ");

    // Service requests at or below the initial head,
    // in descending order.
    for (i = l - 1; i >= 0; i--) {
        total += abs(current - left[i]);
        current = left[i];
        printf("%d ", current);
    }

    // Continue to the end of the disk.
    total += current;
    current = 0;

    // Reverse direction and service remaining requests.
    for (i = 0; i < r; i++) {
        total += abs(right[i] - current);
        current = right[i];
        printf("%d ", current);
    }

    printf("\nTotal head movement = %d\n", total);

    // ---------------- C-LOOK ----------------

    printf("\nC-LOOK (moving toward cylinder 0 first):\n");

    current = head;
    total = 0;

    printf("Service order: ");

    // Service requests at or below the initial head,
    // in descending order.
    for (i = l - 1; i >= 0; i--) {
        total += abs(current - left[i]);
        current = left[i];
        printf("%d ", current);
    }

    // Circular jump to the highest pending request.
    if (r > 0) {
        total += abs(current - right[r - 1]);
        current = right[r - 1];
        printf("%d ", current);

        // Continue servicing in descending order.
        for (i = r - 2; i >= 0; i--) {
            total += abs(current - right[i]);
            current = right[i];
            printf("%d ", current);
        }
    }

    printf("\nTotal head movement = %d\n", total);

    return 0;
}
