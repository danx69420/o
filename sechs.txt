#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>

#define FIFO1 "fifo1"
#define FIFO2 "fifo2"

int main() {
    char buffer[1024];
    int fd1, fd2;
    ssize_t n;

    mkfifo(FIFO1, 0666);
    mkfifo(FIFO2, 0666);

    printf("Enter sentences (type END to finish):\n");

    fd1 = open(FIFO1, O_WRONLY);

    while (fgets(buffer, sizeof(buffer), stdin)) {
        if (strcmp(buffer, "END\n") == 0 ||
            strcmp(buffer, "END") == 0)
            break;

        if (write(fd1, buffer, strlen(buffer)) == -1) {
            perror("Write failed");
            close(fd1);
            return 1;
        }
    }

    close(fd1);

    fd2 = open(FIFO2, O_RDONLY);

    printf("\nResult received from Process 2:\n\n");

    while ((n = read(fd2, buffer, sizeof(buffer))) > 0) {
        write(STDOUT_FILENO, buffer, n);
    }

    close(fd2);

    unlink(FIFO1);
    unlink(FIFO2);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>

#define FIFO1 "fifo1"
#define FIFO2 "fifo2"

int main() {
    char buffer[1024];
    int fd1, fd2;
    FILE *fp;
    ssize_t n;

    long characters = 0;
    long words = 0;
    long lines = 0;
    int in_word = 0;

    mkfifo(FIFO1, 0666);
    mkfifo(FIFO2, 0666);

    fd1 = open(FIFO1, O_RDONLY);

    while ((n = read(fd1, buffer, sizeof(buffer))) > 0) {
        for (ssize_t i = 0; i < n; i++) {
            unsigned char ch = (unsigned char)buffer[i];

            characters++;

            if (ch == '\n')
                lines++;

            if (isspace(ch)) {
                in_word = 0;
            } else if (!in_word) {
                words++;
                in_word = 1;
            }
        }
    }

    close(fd1);

    fp = fopen("output.txt", "w");

    if (fp == NULL) {
        perror("Unable to create output file");
        return 1;
    }

    fprintf(fp, "Number of characters: %ld\n", characters);
    fprintf(fp, "Number of words: %ld\n", words);
    fprintf(fp, "Number of lines: %ld\n", lines);

    fclose(fp);

    fp = fopen("output.txt", "r");

    if (fp == NULL) {
        perror("Unable to open output file");
        return 1;
    }

    fd2 = open(FIFO2, O_WRONLY);

    while ((n = fread(buffer, 1, sizeof(buffer), fp)) > 0) {
        if (write(fd2, buffer, n) == -1) {
            perror("Write failed");
            fclose(fp);
            close(fd2);
            return 1;
        }
    }

    fclose(fp);
    close(fd2);

    printf("Analysis completed. Results saved in output.txt\n");

    return 0;
}
