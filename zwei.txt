#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/types.h>

void bubbleSort(int arr[], int n)
{
    int i, j, temp;

    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - i - 1; j++)
        {
            if (arr[j] > arr[j + 1])
            {
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

void display(int arr[], int n)
{
    int i;

    for (i = 0; i < n; i++)
        printf("%d ", arr[i]);

    printf("\n");
}

int main()
{
    int n, i;
    int arr[100];

    printf("Enter number of integers: ");
    scanf("%d", &n);

    printf("Enter %d integers:\n", n);

    for (i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    // ZOMBIE PROCESS DEMONSTRATION

    pid_t pid = fork();

    if (pid < 0)
    {
        perror("fork failed");
        exit(1);
    }

    if (pid == 0)
    {
        printf("\n--- Child Process (Zombie Demo) ---\n");
        printf("Child PID  : %d\n", getpid());
        printf("Parent PID : %d\n", getppid());

        bubbleSort(arr, n);

        printf("Child sorted array: ");
        display(arr, n);

        printf("Child is terminating...\n");
        exit(0);
    }
    else
    {
        printf("\n--- Parent Process (Zombie Demo) ---\n");
        printf("Parent PID : %d\n", getpid());
        printf("Child PID  : %d\n", pid);

        bubbleSort(arr, n);

        printf("Parent sorted array: ");
        display(arr, n);

        printf("\nParent sleeping for 10 seconds without wait()...\n");
        printf("Child may become a ZOMBIE during this time.\n");

        sleep(10);

        wait(NULL);

        printf("Zombie process cleared using wait().\n");
    }

    // ORPHAN PROCESS DEMONSTRATION

    pid_t pid2 = fork();

    if (pid2 < 0)
    {
        perror("fork failed");
        exit(1);
    }

    if (pid2 == 0)
    {
        printf("\n--- Child Process (Orphan Demo) ---\n");
        printf("Child PID        : %d\n", getpid());
        printf("Original Parent  : %d\n", getppid());

        printf("Child sleeping for 10 seconds...\n");
        sleep(10);

        printf("New Parent PID   : %d\n", getppid());
        printf("Child is still running as an ORPHAN.\n");

        exit(0);
    }
    else
    {
        printf("\n--- Parent Process (Orphan Demo) ---\n");
        printf("Parent PID : %d\n", getpid());
        printf("Child PID  : %d\n", pid2);

        printf("Parent is exiting before the child...\n");
        exit(0);
    }

    return 0;
}
