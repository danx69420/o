#!/bin/bash

while true
do
    echo "----- ADDRESS BOOK -----"
    echo "1. Create Address Book"
    echo "2. View Address Book"
    echo "3. Insert Record"
    echo "4. Delete Record"
    echo "5. Modify Record"
    echo "6. Exit"
    echo "------------------------"

    read -p "Enter your choice: " choice

    case $choice in
    1)
        read -p "Enter file name: " file
        echo -e "NAME\tPHONE\tADDRESS\tBRANCH\tCGPA\tHOSTELLITE?(Y/N)" > "$file"
        echo "Address book created."

        read -p "Enter name: " name
        read -p "Enter phone: " phone
        read -p "Enter address: " address
        read -p "Enter branch: " branch
        read -p "Enter CGPA: " cgpa
        read -p "Hostellite or not? (Y/N): " hostel

        echo -e "$name\t$phone\t$address\t$branch\t$cgpa\t$hostel" >> "$file"
        ;;
    2)
        if [ -f "$file" ]; then
            cat "$file"
        else
            echo "Address book does not exist."
        fi
        ;;
    3)
        read -p "Enter name: " name
        read -p "Enter phone: " phone
        read -p "Enter address: " address
        read -p "Enter branch: " branch
        read -p "Enter CGPA: " cgpa
        read -p "Hostellite or not? (Y/N): " hostel

        echo -e "$name\t$phone\t$address\t$branch\t$cgpa\t$hostel" >> "$file"
        echo "Record inserted."
        ;;
    4)
        read -p "Enter name to delete: " name

        if [ -f "$file" ]; then
            grep -v "^$name[[:space:]]" "$file" > temp
            mv temp "$file"
            echo "Record deleted."
        else
            echo "Address book does not exist."
        fi
        ;;
    5)
        read -p "Enter name to modify: " name

        if [ -f "$file" ]; then
            grep -v "^$name[[:space:]]" "$file" > temp
            mv temp "$file"

            read -p "Enter new name: " newname
            read -p "Enter new phone: " phone
            read -p "Enter new address: " address
            read -p "Enter new branch: " branch
            read -p "Enter new CGPA: " cgpa
            read -p "Hostellite or not? (Y/N): " hostel

            echo -e "$newname\t$phone\t$address\t$branch\t$cgpa\t$hostel" >> "$file"
            echo "Record modified."
        else
            echo "Address book does not exist."
        fi
        ;;
    6)
        echo "Exiting..."
        break
        ;;
    *)
        echo "Invalid choice."
        ;;
    esac
done
